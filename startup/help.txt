SG_ID=$(aws ec2 describe-instances --query 'Reservations[*].Instances[*].SecurityGroups[0].GroupId' --output text)
# Open Port 3000 (Grafana)
aws ec2 authorize-security-group-ingress --group-id $SG_ID --protocol tcp --port 3000 --cidr 0.0.0.0/0
# Open Port 3306 (MySQL)
aws ec2 authorize-security-group-ingress --group-id $SG_ID --protocol tcp --port 3306 --cidr 0.0.0.0/0




# Allow Grafana
sudo ufw allow 3000/tcp
# Allow MySQL
sudo ufw allow 3306/tcp
# Reload firewall
